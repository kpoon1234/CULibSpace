CREATE FUNCTION public.get_table_free_slots(library_open time without time zone DEFAULT '08:00:00'::time without time zone, library_close time without time zone DEFAULT '20:00:00'::time without time zone, p_date date DEFAULT CURRENT_DATE, p_now timestamp without time zone DEFAULT now()) RETURNS TABLE(table_id integer, zone_type character varying, number_of_seat integer, plug_cap integer, free_start timestamp without time zone, free_end timestamp without time zone, duration_min integer)
    LANGUAGE plpgsql
    AS $$
DECLARE
    v_day_open  TIMESTAMP;
    v_day_close TIMESTAMP;
    v_now       TIMESTAMP := p_now;
    v_slot_from TIMESTAMP;
BEGIN
    v_day_open  := p_date + library_open;
    v_day_close := p_date + library_close;

    IF v_day_close <= v_now AND p_date = CURRENT_DATE THEN
        RETURN;
    END IF;

    v_slot_from := CASE
        WHEN p_date = CURRENT_DATE THEN GREATEST(v_day_open, v_now)
        ELSE v_day_open
    END;

    RETURN QUERY
    WITH
    all_tables AS (
        SELECT t.TableID, z.Zone_type, t.Number_of_seat, COALESCE(t.Plug_cap, 0) AS Plug_cap
        FROM  "table" t JOIN zone z ON z.ZoneID = t.ZoneID
        WHERE t.Status != 'Closed'
    ),
    boundaries AS (
        SELECT at.TableID, v_slot_from AS ts FROM all_tables at
        UNION
        SELECT b.TableID, GREATEST(b.StartDateTime, v_slot_from)
        FROM booking b JOIN all_tables at ON at.TableID = b.TableID
        WHERE b.Status IN ('Pending', 'Active')
          AND b.StartDateTime < v_day_close AND b.EndDateTime > v_slot_from
        UNION
        SELECT b.TableID, GREATEST(b.EndDateTime, v_slot_from)
        FROM booking b JOIN all_tables at ON at.TableID = b.TableID
        WHERE b.Status IN ('Pending', 'Active')
          AND b.StartDateTime < v_day_close AND b.EndDateTime > v_slot_from
        UNION
        SELECT at.TableID, v_day_close FROM all_tables at
    ),
    boundary_pairs AS (
        SELECT TableID, ts AS slot_start,
               LEAD(ts) OVER (PARTITION BY TableID ORDER BY ts) AS slot_end
        FROM boundaries
    ),
    valid_slots AS (
        SELECT bp.*
        FROM boundary_pairs bp
        WHERE bp.slot_end IS NOT NULL
          AND bp.slot_start <  v_day_close
          AND bp.slot_end   <= v_day_close
          AND (bp.slot_end - bp.slot_start) >= INTERVAL '30 minutes'
          AND NOT EXISTS (
              SELECT 1 FROM booking b
              WHERE b.TableID = bp.TableID
                AND b.Status IN ('Pending', 'Active')
                AND b.StartDateTime < bp.slot_end
                AND b.EndDateTime   > bp.slot_start
          )
    )
    SELECT
        at.TableID, at.Zone_type, at.Number_of_seat, at.Plug_cap,
        vs.slot_start AS free_start,
        vs.slot_end   AS free_end,
        EXTRACT(EPOCH FROM (vs.slot_end - vs.slot_start))::INT / 60 AS duration_min
    FROM valid_slots vs
    JOIN all_tables at ON at.TableID = vs.TableID
    WHERE (
        p_date + library_close <= p_now
        OR vs.slot_end > (
            DATE_TRUNC('hour', v_now) +
            INTERVAL '30 minutes' * CEIL(EXTRACT(MINUTE FROM v_now) / 30.0)
        )
    )
    ORDER BY at.Zone_type, at.TableID, vs.slot_start;
END;
$$;

CREATE FUNCTION public.get_zone_availability(p_start timestamp without time zone, p_end timestamp without time zone) RETURNS TABLE(zone_id integer, zone_type character varying, total_tables bigint, reserved_tables bigint, available_tables bigint)
    LANGUAGE plpgsql
    AS $$
BEGIN
    RETURN QUERY
    SELECT
        Z.ZoneID,
        Z.Zone_type,
        COUNT(DISTINCT T.TableID) AS total_tables,
        COUNT(DISTINCT B.TableID) AS reserved_tables,
        COUNT(DISTINCT T.TableID) - COUNT(DISTINCT B.TableID) AS available_tables
    FROM ZONE Z
    -- แก้ไขตรงนี้: เพิ่ม AND T.Status != 'Closed' เพื่อไม่นับโต๊ะที่ปิดใช้งาน
    JOIN "table" T ON T.ZoneID = Z.ZoneID AND T.Status != 'Closed'
    LEFT JOIN BOOKING B
        ON B.TableID = T.TableID
        AND B.Status IN ('Pending', 'Active')
        AND B.StartDateTime < p_end
        AND B.EndDateTime > p_start
    GROUP BY Z.ZoneID, Z.Zone_type
    ORDER BY Z.ZoneID;
END;
$$;

CREATE PROCEDURE public.sp_checkinuser(IN p_booking_id integer, IN p_arrive_time timestamp without time zone)
    LANGUAGE plpgsql
    AS $$
DECLARE
    v_start_time      TIMESTAMP;
    v_uid             INT;
    v_table_id        INT;
    v_current_status  VARCHAR(20);
BEGIN
    SELECT StartDateTime, UID, TableID, Status
    INTO v_start_time, v_uid, v_table_id, v_current_status
    FROM BOOKING
    WHERE BookingID = p_booking_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Booking record not found for ID %', p_booking_id;
    END IF;

    IF v_current_status != 'Pending' THEN
        RAISE EXCEPTION 'Check-in failed: Booking is currently % and cannot be processed', v_current_status;
    END IF;

    IF p_arrive_time < (v_start_time - INTERVAL '15 minutes') THEN
        RAISE EXCEPTION 'Check-in failed: You are too early. You can only check in up to 15 minutes before your reservation time.';
    END IF;

    IF p_arrive_time > (v_start_time + INTERVAL '15 minutes') THEN
        UPDATE BOOKING
        SET Status = 'No-show', Arrive_time = p_arrive_time
        WHERE BookingID = p_booking_id;

        UPDATE "table" SET Status = 'Available' WHERE TableID = v_table_id;

        UPDATE "user"
        SET Behaviour_Score = GREATEST(0, Behaviour_Score - 10)
        WHERE UID = v_uid;

        RAISE NOTICE 'Late check-in exceeding 15 minutes: Status updated to No-show and behaviour score deducted successfully.';
    ELSE
        UPDATE BOOKING
        SET Status = 'Active', Arrive_time = p_arrive_time
        WHERE BookingID = p_booking_id;

        UPDATE "table" SET Status = 'Occupied' WHERE TableID = v_table_id;

        RAISE NOTICE 'Check-in successful: Welcome to the Central Library space.';
    END IF;
END;
$$;

--

CREATE PROCEDURE public.sp_reservetable(IN p_uid integer, IN p_table_id integer, IN p_start_time timestamp without time zone, IN p_end_time timestamp without time zone)
    LANGUAGE plpgsql
    AS $$
DECLARE
    v_user_score        DECIMAL(4,1);
    v_overlapping_count INT;
    v_table_status      VARCHAR(20);
BEGIN
    IF p_start_time < CURRENT_TIMESTAMP THEN
        RAISE EXCEPTION 'Booking failed: Cannot reserve a table in the past';
    END IF;

    SELECT Behaviour_Score INTO v_user_score FROM "user" WHERE UID = p_uid;
    IF v_user_score < 50 THEN
        RAISE EXCEPTION 'Booking failed: Your behaviour score (%) is below the required threshold', v_user_score;
    END IF;

    SELECT Status INTO v_table_status FROM "table" WHERE TableID = p_table_id;
    IF v_table_status = 'Closed' THEN
        RAISE EXCEPTION 'Booking failed: Table % is currently closed for maintenance', p_table_id;
    END IF;

    SELECT COUNT(*) INTO v_overlapping_count
    FROM BOOKING
    WHERE TableID = p_table_id
      AND Status IN ('Pending', 'Active')
      AND (p_start_time, p_end_time) OVERLAPS (StartDateTime, EndDateTime);

    IF v_overlapping_count > 0 THEN
        RAISE EXCEPTION 'Booking failed: Table % is already booked during the requested time', p_table_id;
    END IF;

    INSERT INTO BOOKING (UID, TableID, StartDateTime, EndDateTime, Status, Timestamp)
    VALUES (p_uid, p_table_id, p_start_time, p_end_time, 'Pending', CURRENT_TIMESTAMP);

    RAISE NOTICE 'Booking successful: Table % from % to %', p_table_id, p_start_time, p_end_time;
END;
$$;
